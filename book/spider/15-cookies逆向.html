cookie加密

Cookie反爬 或 动态Cookie验证 的反爬虫技术 ==》 成为加速乐

注意事项

  1. 清空cookie
  2. 勾选上记录所有请求信息Preserve log
  3. 选择只展示Doucment文件
  4. 涮新则可以查看数据

示例

案例地址： https://www.cnvd.org.cn/

流程:

  1. 第一次请求: 获取后端服务器返回的__jsluid_s的cookie
    • 获取后端js代码,用于生成对应cookie
    • 下次请求时将两个cookie携带到cookie中
  2. 第二次请求时: 携带第一次请求返回的cookie和js生成的cookie
    • 获取一个混淆的js代码(md5 sha1 sha256等等)用于生成cookie
  3. 第三次请求: 携带第二次生成的cookie

此处案例: 需要还需要进行验证码验证

  • 验证码通过之后,会返回一个新的cookie,获取新的cookie添加到对应接口即可获取到数据
  • 或者到新cookie之后: __jsluid_s和__jsl_clearance_scookie不需要变动了,只需要更改携带参数即可获取到分页数据
import time
import requests
import re
import execjs
import base64
from lxml import etree


class SpiderCookie:
    def __init__(self):
        self.headers = {
            "accept": "text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8,application/signed-exchange;v=b3;q=0.7",
            "accept-language": "zh-CN,zh;q=0.9",
            "priority": "u=0, i",
            "referer": "https://www.cnvd.org.cn/flaw/list?numPerPage=10&offset=10&keywordFlag=0&cnvdIdFlag=0&baseinfoBeanFlag=0&max=10",
            "sec-ch-ua": "\"Chromium\";v=\"148\", \"Google Chrome\";v=\"148\", \"Not/A)Brand\";v=\"99\"",
            "sec-ch-ua-mobile": "?0",
            "sec-ch-ua-platform": "\"Windows\"",
            "sec-fetch-dest": "document",
            "sec-fetch-mode": "navigate",
            "sec-fetch-site": "same-origin",
            "sec-fetch-user": "?1",
            "upgrade-insecure-requests": "1",
            "user-agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/148.0.0.0 Safari/537.36"
        }
        # 页面数据地址
        self.url = "https://www.cnvd.org.cn/flaw/list"
        # 携带的cookie
        self.cookies = {
            "__jsluid_s": "",
            "__jsl_clearance_s": ""
        }
        # 翻页数据
        self.params = {
            "numPerPage": "10",
            "offset": "10",
            "keywordFlag": "0",
            "cnvdIdFlag": "0",
            "baseinfoBeanFlag": "0",
            "max": "10"
        }

    # 第一次请求: 获取后端返回的cookie, 后端返回的js代码=>执行之后获取的cookie值
    def one_spider(self):
        response = requests.get(self.url, headers=self.headers, params=self.params)
        # 用于将响应cookie转换为字典的方法
        __jsluid_s = response.cookies.get_dict()["__jsluid_s"]
        # 获取到一段js代码  需要再python执行js代码
        jsl_result = re.findall(r'document.cookie=(.*);location', response.text)[0]
        # 通过execjs.eval直接执行js表达式代码 进行切割获取数据
        __jsl_clearance_s = execjs.eval(jsl_result).split(";")[0].split("=")[1]
        print("第一次请求", __jsluid_s, __jsl_clearance_s)
        self.cookies = {
            "__jsluid_s": __jsluid_s,
            "__jsl_clearance_s": __jsl_clearance_s
        }

    # 第二次请求: 将第一次请求的cookie和js生成的cookie携带上进行请求
    def two_spider(self):
        # 获取到一段混淆的代码 通过正在则进行处理数据
        response = requests.get(self.url, headers=self.headers, params=self.params, cookies=self.cookies)
        with open('./cookie/cookie1.js', 'w', encoding="utf-8") as f:
            f.write(response.text)
        # 通过正则处理数据 去除script标签
        match = re.search(r'<script[^>]*>([\s\S]*?)</script>', response.text)
        modified_js = match.group(1)
        # 不需要去除定时器 直接执行就可以 需要拼接内容
        env_code = """window = {};
        navigator = {
            "userAgent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/148.0.0.0 Safari/537.36"
        }
        document = {};
        location = {
            "href": "",
        };
        window.navigator = navigator;
        var setTimeout = function(callback, delay) {
            callback();
        };

        """
        tail_code = """
        function getCookie() {
            console.log(document.cookie)
            return document.cookie;
        }
        """
        final_js = env_code + modified_js + tail_code
        # 写入js文件中  其实写不写入  无所谓
        with open('./cookie/cookie.js', 'w', encoding="utf-8") as f:
            f.write(final_js)
        # with open('./cookie/cookie.js', 'r', encoding="utf-8") as fs:
        #     result = fs.read()
        ctx = execjs.compile(final_js)
        msg = ctx.call("getCookie")
        __jsl_clearance_s = msg.split(";")[0].split("=")[1]
        self.cookies["__jsl_clearance_s"] = __jsl_clearance_s
        print("第二次请求", self.cookies)

    # 第三次请求: 直接获取文档地址是没有办法获取到数据, 需要通过验证码验证
    # 正常加速乐cookie: 第三次是获取数据的,由于这个是需要通过验证,所以需要多次处理
    def three_spider(self):
        # 图片验证码
        url_img = "https://www.cnvd.org.cn/cdn-cgi/captcha/v2/captcha/image"
        # 需要携带上次的cookie进行获取图片
        response = requests.get(url_img, headers=self.headers, cookies=self.cookies)
        # 获取图片地址  进行图片转换
        data_img_info = response.json()
        print(data_img_info["image"], "图片数据")
        print(data_img_info["sec"], "数据")
        # image_data = base64.b64decode(data_img_info["image"])
        # with open("output.png", "wb") as f:
        #     f.write(image_data)
        # with open('output.png', 'rb') as f:
        #     b = base64.b64encode(f.read()).decode()  ## 图片二进制流base64字符串
        # 获取图片验证码
        url = "http://api.jfbym.com/api/YmServer/customApi"
        data = {
            ## 关于参数,一般来说有3个;不同类型id可能有不同的参数个数和参数名,找客服获取
            "token": "RJYp5oyWpTsacY3dlxYlxKh61ekP1XpiIZfwJU6sFUM",
            "type": "50100",
            "image": data_img_info["image"],
        }
        _headers = {
            "Content-Type": "application/json"
        }
        response = requests.request("POST", url, headers=_headers, json=data).json()
        print(response['data']["data"], "验证")

        # 验证码进行验证
        url_yan_zhen = "https://www.cnvd.org.cn/cdn-cgi/captcha/v2/captcha/image"
        data = {
            "ans": response['data']["data"],
            "sec": data_img_info["sec"]
        }
        # 图片验证
        response = requests.post(url_yan_zhen, headers=self.headers, cookies=self.cookies, data=data)
        print("图片验证", response.text)
        __jsl_clearance_s = response.cookies.get_dict()["__jsl_clearance_s"]
        self.cookies["__jsl_clearance_s"] = __jsl_clearance_s

    def result_spider(self):
        for i in range(1, 4):
            params = {
                "numPerPage": "10",
                "offset": i * 10,
                "keywordFlag": "0",
                "cnvdIdFlag": "0",
                "baseinfoBeanFlag": "0",
                "max": "10"
            }
            response = requests.get(self.url, headers=self.headers, params=params, cookies=self.cookies)
            xml = etree.HTML(response.text)
            titles = xml.xpath("//table[@class='tlist']//tr/td/a/@title")
            print(titles, "数据")
            time.sleep(3)
        pass


if __name__ == '__main__':
    spider = SpiderCookie()
    spider.one_spider()
    spider.two_spider()
    spider.three_spider()
    spider.result_spider()
