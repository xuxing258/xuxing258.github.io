OCR识别字体文件

安装以下四个库

  • fontTools 处理字体文件库
  • ddddocr 通用验证码离线本地识别库
    • 文字识别 (OCR)：识别图片中的文字内容，这是最基础的功能
    • 滑块验证码识别：能够识别滑块验证码中缺口的位置
    • 目标检测 (Detection)：检测图片中特定目标（如验证码字符）的位置
    • 点选验证码识别：可以定位点选验证码中目标的坐标
  • matplotlib 用于将字体轮廓渲染成图片
  • freetype-py 将字体中的字符轮廓（矢量图形）渲染成像素图像
import os
from fontTools.ttLib import TTFont
from fontTools.pens.freetypePen import FreeTypePen
import matplotlib.pyplot as plt
import ddddocr

def get_font_img():
    # 1. 加载字体文件
    font = TTFont('./file/font.woff')  
    # 2. 获取字符映射表 (cmap)
    cmap = font.getBestCmap()
    # 3. 创建一个文件夹来存放渲染出的图片
    os.makedirs('font_images', exist_ok=True)
    # 4. 遍历所有字符，渲染并保存为图片
    for code, glyph_name in cmap.items():
        # 根据字形名称获取字形对象 _TTGlyphGlyf内容
        glyph = font.getGlyphSet()[glyph_name]
        # 使用FreeTypePen将字形“画”出来
        pen = FreeTypePen(None)
        glyph.draw(pen)
        # 将画好的内容转换为图像数组
        img_array = pen.array()
        # 使用matplotlib保存为图片
        plt.figure()
        plt.imshow(img_array, cmap='gray')
        plt.axis('off')
        # 以编码值命名图片，例如 '1040.jpg'
        plt.savefig(f'font_images/{code}.jpg', bbox_inches='tight', pad_inches=0)
        plt.close()
    print(f"渲染{len(cmap)}张图片到 'font_images' 文件夹")
    pass

# 批量识别 font_images 文件夹里所有 .jpg 图片上的字符
def handler_result():
     # 初始化 OCR 对象
    ocr = ddddocr.DdddOcr(show_ad=False)  
    mapping = {}                         
	 # 遍历 font_images 文件夹
    for filename in os.listdir('font_images'): 
        # 只处理 .jpg 文件
        if filename.endswith('.jpg'):   
            # 从文件名提取数字编码（如 97.jpg → 97）
            code = int(filename.split('.')[0])  
            with open(os.path.join('font_images', filename), 'rb') as f:
                #  以二进制方式读取图片内容
                image_bytes = f.read()       
            # 调用 ddddocr 识别图片中的文字
            result = ocr.classification(image_bytes)  
            if result:                        
                mapping[chr(code)] = result   
            else:                            
                print(f"警告: 编码 {code} 识别失败")
    print("\n最终建立的映射字典:", mapping)  
    return mapping

if __name__ == '__main__':
    get_font_img()
    mapping = handler_result()

    new_str = "设多儿儿物系"
    qq_str = ""
    for i in new_str:
        qq_str += str(mapping[i])
    print(qq_str)

import ddddocr
import random

def solve_slide_from_local(bg_path, slide_path):
    """
    bg_path: 本地背景图的路径，例如 './bg.jpg'
    slide_path: 本地滑块小图的路径，例如 './slide.png'
    """
    # 1. 初始化识别器（关闭OCR和检测，只做滑块匹配）
    det = ddddocr.DdddOcr(det=False, ocr=False, show_ad=False)

    # 2. 以二进制方式读取本地图片（这是处理本地文件最直接的方式）
    with open(bg_path, 'rb') as f:
        bg_bytes = f.read()
    
    with open(slide_path, 'rb') as f:
        slide_bytes = f.read()

    # 3. 调用滑块匹配方法
    # 注意：第一个参数是滑块图，第二个参数是背景图
    res = det.slide_match(slide_bytes, bg_bytes, simple_target=True)

    distance = res['target'][0]
    print(f"识别到的缺口距离: {distance} 像素")
