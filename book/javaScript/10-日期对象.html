Date

用于处理日期和时间的内置对象。它提供了许多方法和属性，可以获取和操作日期、时间、年份、月份、小时、分钟、秒等

日期格式化方法

let now = new Date()
console.log(now);         // 本地时间
console.log(now.toDateString());  // 年月日
console.log(now.toTimeString());  // 时分秒
console.log(now.toUTCString());   // 国际标准时间 
console.log(now.toLocaleDateString());  // 年月日
console.log(now.toLocaleString());  // 年月日 时分秒 
console.log(now.toLocaleTimeString()); // 时分秒

let nowT = Date.now() //返回当前事件距离1970年1月1日00:00:00之间的时间戳距离

常用方法

let currentDate = new Date();
console.log(currentDate); 	// 输出当前日期和时间  

// 创建一个指定日期和时间的 Date 对象
let specificDate = new Date(2023, 11, 25, 12, 0, 0);
console.log(specificDate); // 输出指定日期和时间：December 25, 2023 12:00:00

let date = new Date();
console.log(date.getFullYear()); // 获取年份
console.log(date.getMonth()); // 获取月份（0-11，0 表示一月）
console.log(date.getDate()); // 获取日期
console.log(date.getHours()); // 获取小时
console.log(date.getMinutes()); // 获取分钟
console.log(date.getSeconds()); // 获取秒
console.log(date.getMilliseconds()); // 获取毫秒

自定义格式化函数

function formatDate(date) {
  let year = date.getFullYear();
  let month = ("0" + (date.getMonth() + 1)).slice(-2);
  let day = ("0" + date.getDate()).slice(-2);
  let hours = ("0" + date.getHours()).slice(-2);
  let minutes = ("0" + date.getMinutes()).slice(-2);
  let seconds = ("0" + date.getSeconds()).slice(-2);
  
  return year + "-" + month + "-" + day + " " + hours + ":" + minutes + ":" + seconds;
}

let date = new Date();
let formattedDate = formatDate(date);
console.log(formattedDate); // 输出自定义格式的日期字符串，例如："2023-12-18 18:30:00"
